#Define input function for CSV
dialysis_read_csv<-function(x) {
read_csv(x, col_types = cols(Zip = col_character()),
         locale=locale(encoding="latin1"))%>%
    rename_with(toupper)%>%
    rename(TOTSTAS='# OF DIALYSIS STATIONS')%>%
    rename_with(~case_when(
      .=="PROFIT OR NON-PROFIT"~"PROFIT_STATUS",
      .=="PROFIT OR NON-PROFIT?"~"PROFIT_STATUS",
      TRUE~.
    ))%>%
    select(ZIP, TOTSTAS, PROFIT_STATUS)%>%
    mutate(ZIP=case_when(
      str_length(ZIP)==5~ZIP,
      str_length(ZIP)==4~paste0(0,ZIP),
      str_length(ZIP)==3~paste0(0,0,ZIP),
    ))%>%
    group_by(ZIP)%>%
    summarize(TotalDialysisCentersZIP=n(), 
              TotalDialysisStationsZIP=sum(TOTSTAS),
              TotalProfitDialysisCentersZIP=sum(PROFIT_STATUS=="Profit"),
              TotalProfitDialysisStationsZIP=sum(TOTSTAS[PROFIT_STATUS=="Profit"]),
              TotalNonProfitDialysisCentersZIP=sum(PROFIT_STATUS=="Non-Profit"),
              TotNonProfitDialysisStationsZIP=sum(TOTSTAS[PROFIT_STATUS=="Non-Profit"])
              )%>%
    return()
}

#Define input function for TXT
dialysis_read_txt<-function(x) {
  read_delim(x, "\t", escape_double = FALSE, col_types = cols(phyzip = col_character()), trim_ws = TRUE) %>%
    rename_with(toupper)%>%
    rename(ZIP= PHYZIP)%>%
    rename(PROFIT_STATUS=OWNTYPE)%>%
    select(ZIP, TOTSTAS, PROFIT_STATUS)%>%
    mutate(ZIP=case_when(
      str_length(ZIP)==5~ZIP,
      str_length(ZIP)==4~paste0(0,ZIP),
      str_length(ZIP)==3~paste0(0,0,ZIP),
    ))%>%
    group_by(ZIP)%>%
    summarize(TotalDialysisCentersZIP=n(), 
              TotalDialysisStationsZIP=sum(TOTSTAS),
              TotalProfitDialysisCentersZIP=sum(PROFIT_STATUS==1),
              TotalProfitDialysisStationsZIP=sum(TOTSTAS[PROFIT_STATUS==1]),
              TotalNonProfitDialysisCentersZIP=sum(PROFIT_STATUS==2),
              TotNonProfitDialysisStationsZIP=sum(TOTSTAS[PROFIT_STATUS==2])
              )%>%
    return()
}
